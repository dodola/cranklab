cmake_minimum_required(VERSION 3.22)
project(crank_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
option(CRANK_SNAPDRAGON "Build Hexagon HTP + Adreno OpenCL plugins (CPU needs i8mm; use build-snapdragon.sh)" OFF)
if(ANDROID)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  if(CRANK_SNAPDRAGON)
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    set(GGML_HEXAGON ON CACHE BOOL "" FORCE)
  else()
    # Runtime CPU scoring selects DotProd/FP16 only on devices that support it.
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  endif()
  string(APPEND CMAKE_SHARED_LINKER_FLAGS " -Wl,-z,max-page-size=16384")
  # ggml dynamically loads CPU variants as MODULE libraries.
  string(APPEND CMAKE_MODULE_LINKER_FLAGS " -Wl,-z,max-page-size=16384")
endif()
add_subdirectory(deps/llama.cpp)
# Android packages plain .so names; do not create dependencies on .so.0 files.
foreach(target llama ggml ggml-base)
  set_target_properties(${target} PROPERTIES VERSION "" SOVERSION "")
endforeach()
add_library(crank-translate SHARED crank_translate.cpp)
target_link_libraries(crank-translate PRIVATE llama vendor-hash)
set_target_properties(crank-translate PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
