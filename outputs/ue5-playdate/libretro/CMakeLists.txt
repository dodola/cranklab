cmake_minimum_required(VERSION 3.20)
project(CrankLabLibretro LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT UNIX OR APPLE OR ANDROID)
    message(FATAL_ERROR "This libretro build targets desktop and handheld Linux (x86_64, aarch64, armhf)")
endif()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(RT "${CMAKE_CURRENT_SOURCE_DIR}/../runtime")
set(PD_DEPS_ROOT "${RT}/deps")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)

# zlib
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${PD_DEPS_ROOT}/zlib" "${CMAKE_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)

# Unicorn, ARM only. QEMU's configure script runs on the build machine.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UNICORN_ARCH "arm" CACHE STRING "Guest CPU architecture" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${PD_DEPS_ROOT}/unicorn" "${CMAKE_BINARY_DIR}/unicorn")
get_property(PD_UNICORN_TARGETS DIRECTORY "${PD_DEPS_ROOT}/unicorn" PROPERTY BUILDSYSTEM_TARGETS)
foreach(uc_target ${PD_UNICORN_TARGETS})
    get_target_property(uc_type ${uc_target} TYPE)
    if(uc_type MATCHES "^(STATIC|OBJECT|SHARED)_LIBRARY$")
        target_compile_options(${uc_target} PRIVATE "SHELL:-include \"${RT}/compat/unicorn_crc.h\"")
    endif()
endforeach()

# SDL2 and SDL_mixer are built privately, with only the dummy drivers, so the
# core never touches the frontend's own SDL state.
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
foreach(feature X11 WAYLAND KMSDRM RPI VIVANTE OPENGL OPENGLES VULKAN OFFSCREEN DIRECTFB
        ALSA JACK ESD PIPEWIRE PULSEAUDIO ARTS NAS SNDIO OSS DISKAUDIO LIBSAMPLERATE
        HIDAPI LIBUDEV DBUS IBUS HAPTIC SENSOR POWER FILESYSTEM LOCALE MISC CCACHE)
    set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL_RENDER_D3D OFF CACHE BOOL "" FORCE)
set(SDL_DLOPEN OFF CACHE BOOL "" FORCE)
set(SDL_VIDEO ON CACHE BOOL "" FORCE)
set(SDL_RENDER ON CACHE BOOL "" FORCE)
set(SDL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_RPATH OFF CACHE BOOL "" FORCE)
add_subdirectory("${PD_DEPS_ROOT}/sdl-source" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
target_compile_definitions(SDL2-static PRIVATE CRANK_LOCKSTEP_AUDIO)
foreach(feature SAMPLES VENDORED CMD MOD MIDI OPUS WAVPACK)
    set(SDL2MIXER_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${PD_DEPS_ROOT}/mixer-source" "${CMAKE_BINARY_DIR}/mixer" EXCLUDE_FROM_ALL)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/compat-include/SDL2")
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/compat-include/SDL2/SDL.h" CONTENT "#include <SDL.h>\n" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/compat-include/SDL2/SDL_mixer.h" CONTENT "#include <SDL_mixer.h>\n" @ONLY)

file(GLOB LUA_SOURCE CONFIGURE_DEPENDS "${RT}/vendor/lua/*.c")
file(GLOB RUNTIME_SOURCE CONFIGURE_DEPENDS "${RT}/src/*.c")
list(FILTER RUNTIME_SOURCE EXCLUDE REGEX "pd_http_(win32|android)\\.c$")
list(APPEND RUNTIME_SOURCE "${RT}/src/pd_native.cpp")

add_library(cranklab_libretro SHARED ${LUA_SOURCE} ${RUNTIME_SOURCE} src/cranklab_libretro.c src/cranklab_content.c src/cranklab_menu.c)
set_target_properties(cranklab_libretro PROPERTIES PREFIX "" LINKER_LANGUAGE CXX)
target_compile_definitions(cranklab_libretro PRIVATE PD_LIBRARY LUA_32BITS SDL_MAIN_HANDLED CRANK_EMBEDDED_SDL CRANK_LOCKSTEP_AUDIO _GNU_SOURCE)
target_compile_options(cranklab_libretro PRIVATE -include "${RT}/compat/pd_platform.h" -fno-strict-aliasing)
target_include_directories(cranklab_libretro PRIVATE include "${RT}/include" "${RT}/src" "${RT}/vendor/lua"
    "${PD_DEPS_ROOT}/zlib" "${CMAKE_BINARY_DIR}/zlib" "${CMAKE_BINARY_DIR}/compat-include")
target_link_libraries(cranklab_libretro PRIVATE unicorn zlibstatic SDL2_mixer SDL2-static m dl pthread)
target_link_options(cranklab_libretro PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/link.T"
    -Wl,-Bsymbolic -Wl,--gc-sections -static-libstdc++ -static-libgcc)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(cranklab_libretro PRIVATE -Wl,--strip-all)
endif()
