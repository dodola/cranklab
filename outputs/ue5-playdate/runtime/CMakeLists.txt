cmake_minimum_required(VERSION 3.20)
project(CrankRuntime LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PD_DEPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps" CACHE PATH "Directory containing SDL2, mixer and zlib sources")
if(NOT WIN32 AND NOT ANDROID)
    message(FATAL_ERROR "Supported targets are Windows and Android")
endif()
file(GLOB LUA_SOURCE CONFIGURE_DEPENDS vendor/lua/*.c)
file(GLOB RUNTIME_SOURCE CONFIGURE_DEPENDS src/*.c)
list(APPEND RUNTIME_SOURCE src/pd_native.cpp)
list(FILTER RUNTIME_SOURCE EXCLUDE REGEX "pd_http_(win32|android)\\.c$")
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${PD_DEPS_ROOT}/zlib" "${CMAKE_BINARY_DIR}/zlib")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UNICORN_ARCH "arm" CACHE STRING "Guest CPU architecture" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "" FORCE)
if(CMAKE_HOST_WIN32 AND ANDROID)
    # QEMU's feature tests compile for the NDK target, but its configure script
    # runs on the host. Git for Windows supplies sh/sed without a global install.
    find_program(PD_HOST_SH sh HINTS "C:/Program Files/Git/usr/bin")
    if(NOT PD_HOST_SH)
        message(FATAL_ERROR "Android Unicorn configuration needs Git for Windows sh in PATH")
    endif()
    get_filename_component(PD_HOST_SH_DIR "${PD_HOST_SH}" DIRECTORY)
    set(ENV{PATH} "${PD_HOST_SH_DIR};$ENV{PATH}")
    get_filename_component(PD_NDK_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
    set(ENV{STRINGS} "${PD_NDK_BIN}/llvm-strings.exe")
    # This ARM-only Unicorn build uses bundled glib and no pkg-config packages.
    # The legacy configure script still requires an executable probe command.
    set(ENV{PKG_CONFIG} "false")
endif()
add_subdirectory("${PD_DEPS_ROOT}/unicorn" "${CMAKE_BINARY_DIR}/unicorn")
if(NOT MSVC AND NOT EXISTS "${CMAKE_BINARY_DIR}/config-host.h")
    message(FATAL_ERROR "Unicorn configure did not generate config-host.h; see config.log")
endif()
# Unicorn's QEMU CRC implementation has a different ABI from zlib's crc32.
# Give it a private symbol in every Unicorn translation unit; resolving the
# duplicate by link order would risk calling the wrong implementation.
get_property(PD_UNICORN_TARGETS DIRECTORY "${PD_DEPS_ROOT}/unicorn" PROPERTY BUILDSYSTEM_TARGETS)
foreach(uc_target ${PD_UNICORN_TARGETS})
    get_target_property(uc_type ${uc_target} TYPE)
    if(uc_type MATCHES "^(STATIC|OBJECT|SHARED)_LIBRARY$")
        if(MSVC)
            target_compile_options(${uc_target} PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/compat/unicorn_crc.h")
        else()
            target_compile_options(${uc_target} PRIVATE "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/compat/unicorn_crc.h\"")
        endif()
    endif()
endforeach()
if(ANDROID)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
    # SDL's Android JNI source references video/controller declarations unconditionally.
    # The embedded host only initializes timers and the dummy audio output.
    foreach(feature VIDEO RENDER JOYSTICK HAPTIC HIDAPI SENSOR)
        set(SDL_${feature} ON CACHE BOOL "" FORCE)
    endforeach()
    foreach(feature POWER FILESYSTEM LOCALE MISC)
        set(SDL_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${PD_DEPS_ROOT}/sdl-source" "${CMAKE_BINARY_DIR}/sdl")
    target_compile_definitions(SDL2-static PRIVATE CRANK_EMBEDDED_SDL)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    foreach(feature SAMPLES VENDORED CMD MOD MIDI OPUS WAVPACK)
        set(SDL2MIXER_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    # MP3, FLAC, Vorbis and WAVE decoders are included in upstream mixer sources.
    add_subdirectory("${PD_DEPS_ROOT}/mixer-source" "${CMAKE_BINARY_DIR}/mixer")
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/compat-include/SDL2")
    file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/compat-include/SDL2/SDL.h" CONTENT "#include <SDL.h>\n" @ONLY)
    file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/compat-include/SDL2/SDL_mixer.h" CONTENT "#include <SDL_mixer.h>\n" @ONLY)
    add_library(crank-core SHARED ${LUA_SOURCE} ${RUNTIME_SOURCE} src/pd_http_android.c)
    set_target_properties(crank-core PROPERTIES LINKER_LANGUAGE CXX C_VISIBILITY_PRESET hidden)
    set(PD_TARGETS crank-core)
else()
    add_executable(crank-runtime ${LUA_SOURCE} ${RUNTIME_SOURCE} src/pd_http_win32.c)
    add_library(crank-core SHARED ${LUA_SOURCE} ${RUNTIME_SOURCE} src/pd_http_win32.c)
    set(PD_TARGETS crank-runtime crank-core)
endif()
target_compile_definitions(crank-core PRIVATE PD_LIBRARY)
foreach(target ${PD_TARGETS})
    target_compile_definitions(${target} PRIVATE LUA_32BITS SDL_MAIN_HANDLED _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
    target_compile_definitions(${target} PRIVATE NOMINMAX)
    target_link_libraries(${target} PRIVATE unicorn)
    target_include_directories(${target} PRIVATE include src vendor/lua "${PD_DEPS_ROOT}/zlib" "${CMAKE_BINARY_DIR}/zlib")
    if(WIN32)
        target_compile_options(${target} PRIVATE /utf-8 /FI"${CMAKE_CURRENT_SOURCE_DIR}/compat/pd_platform.h")
        target_include_directories(${target} PRIVATE compat "${PD_DEPS_ROOT}/include")
        target_link_libraries(${target} PRIVATE zlibstatic winhttp "${PD_DEPS_ROOT}/lib/SDL2.lib" "${PD_DEPS_ROOT}/lib/SDL2_mixer.lib")
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PD_DEPS_ROOT}/lib/SDL2.dll" "$<TARGET_FILE_DIR:${target}>"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PD_DEPS_ROOT}/lib/SDL2_mixer.dll" "$<TARGET_FILE_DIR:${target}>")
    else()
        target_compile_options(${target} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat/pd_platform.h")
        target_include_directories(${target} PRIVATE "${CMAKE_BINARY_DIR}/compat-include")
        target_link_libraries(${target} PRIVATE zlibstatic SDL2_mixer SDL2-static log android)
        target_link_options(${target} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
    endif()
endforeach()
if(ANDROID)
    add_executable(crank-probe ../tools/android-probe.c)
    target_include_directories(crank-probe PRIVATE include)
    target_link_libraries(crank-probe PRIVATE crank-core)
    target_link_options(crank-probe PRIVATE "-Wl,-z,max-page-size=16384")
endif()
