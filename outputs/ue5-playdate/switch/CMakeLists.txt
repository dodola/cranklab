cmake_minimum_required(VERSION 3.20)
project(PlaydateSwitch VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
if(NINTENDO_SWITCH)
    set(PKG_CONFIG_ARGN --static)
endif()
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
pkg_check_modules(TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(ZLIB REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(RT "${CMAKE_CURRENT_SOURCE_DIR}/../runtime")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
add_custom_command(OUTPUT "${GENERATED}/pd_launcher_switch.inc" "${GENERATED}/theme.h"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate.py" --output "${GENERATED}"
    DEPENDS launcher/pdgames.lua tools/generate.py ../Unreal/Source/CrankLab/CrankTheme.h VERBATIM)
file(GLOB LUA CONFIGURE_DEPENDS "${RT}/vendor/lua/*.c")
file(GLOB CORE CONFIGURE_DEPENDS "${RT}/src/*.c")
list(FILTER CORE EXCLUDE REGEX "pd_http_(win32|android|stub)\\.c$")
add_library(crank-switch STATIC ${LUA} ${CORE} src/lua_only.c src/http.c "${GENERATED}/pd_launcher_switch.inc")
target_compile_definitions(crank-switch PRIVATE PD_LIBRARY LUA_32BITS SDL_MAIN_HANDLED CRANK_EXTERNAL_SDL
    PD_LAUNCHER_SCRIPT_INCLUDE="pd_launcher_switch.inc" _GNU_SOURCE)
target_compile_options(crank-switch PRIVATE "SHELL:-include \"${RT}/compat/pd_platform.h\"" -fno-strict-aliasing)
if(NINTENDO_SWITCH)
    target_compile_options(crank-switch PRIVATE "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/src/platform.h\"")
endif()
target_include_directories(crank-switch PUBLIC "${RT}/include" PRIVATE "${RT}/src" "${RT}/vendor/lua" "${GENERATED}" src)
target_link_libraries(crank-switch PUBLIC PkgConfig::SDL2 PkgConfig::MIXER PkgConfig::CURL ZLIB::ZLIB m)
add_library(switch-services STATIC src/archive.cpp src/launcher.cpp src/library.cpp)
target_link_libraries(switch-services PUBLIC crank-switch)
target_include_directories(switch-services PUBLIC src)
target_compile_options(switch-services PRIVATE -Wall -Wextra)
add_executable(playdate-switch src/main.cpp src/input.cpp ../libretro/src/cranklab_menu.c "${GENERATED}/theme.h")
target_include_directories(playdate-switch PRIVATE src "${GENERATED}" ../libretro/src "${RT}/src" "${RT}/vendor/lua")
target_link_libraries(playdate-switch PRIVATE switch-services PkgConfig::TTF)
target_compile_options(playdate-switch PRIVATE -Wall -Wextra)
if(NINTENDO_SWITCH)
    target_compile_definitions(crank-switch PRIVATE __SWITCH__)
    target_compile_definitions(playdate-switch PRIVATE __SWITCH__)
    nx_generate_nacp(playdate-switch.nacp NAME "Playdate" AUTHOR "CrankLab" VERSION "${PROJECT_VERSION}")
    dkp_add_asset_target(playdate-romfs "${CMAKE_CURRENT_SOURCE_DIR}/romfs")
    file(GLOB_RECURSE ROMFS_FILES CONFIGURE_DEPENDS LIST_DIRECTORIES true "${CMAKE_CURRENT_SOURCE_DIR}/romfs/*")
    set_property(TARGET playdate-romfs PROPERTY DKP_ASSET_FILES "${ROMFS_FILES};${CMAKE_CURRENT_SOURCE_DIR}/romfs")
    nx_create_nro(playdate-switch NACP playdate-switch.nacp ROMFS playdate-romfs ICON "${CMAKE_CURRENT_SOURCE_DIR}/assets/icon.jpg")
else()
    enable_testing()
    add_library(crank-switch-probe SHARED src/probe_anchor.c)
    target_link_libraries(crank-switch-probe PRIVATE -Wl,--whole-archive crank-switch -Wl,--no-whole-archive)
    add_executable(core-test tests/core_test.cpp src/library.cpp src/input.cpp)
    target_include_directories(core-test PRIVATE src)
    target_link_libraries(core-test PRIVATE crank-switch)
    add_test(NAME core COMMAND core-test "${CMAKE_CURRENT_BINARY_DIR}/test-data")
    set_tests_properties(core PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy;SDL_VIDEODRIVER=dummy")
    add_executable(services-test tests/services_test.cpp)
    target_link_libraries(services-test PRIVATE switch-services)
    add_test(NAME services COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_services.py"
        --executable "$<TARGET_FILE:services-test>" --output "${CMAKE_CURRENT_BINARY_DIR}/service-test-data")
    set_tests_properties(services PROPERTIES TIMEOUT 60)
endif()
